# 首版仅支持项目内 GCC 16.2 / Linux, 缺少能力时明确失败, 不降级标准或下载替代工具.
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_CXX_COMPILER_ID STREQUAL "GNU" OR NOT CMAKE_CXX_COMPILER_VERSION VERSION_EQUAL 16.2.0)
    message(FATAL_ERROR "Star/Planet C++ requires Linux and the pinned GCC 16.2.0 toolchain")
endif()
set(CMAKE_CXX_STANDARD 26)

# 开发产物可直接运行. 安装阶段不继承这个开发机路径, 发布时另行提供匹配运行库.
execute_process(COMMAND "${CMAKE_CXX_COMPILER}" -print-file-name=libstdc++.so.6
    OUTPUT_VARIABLE star_cxx_runtime OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
get_filename_component(star_cxx_runtime_dir "${star_cxx_runtime}" DIRECTORY)
set(CMAKE_BUILD_RPATH "${star_cxx_runtime_dir}")
set(FETCHCONTENT_FULLY_DISCONNECTED ON CACHE BOOL "No implicit downloads" FORCE)
option(ASTRA_CORE_ONLY "Build only offline configuration and role-state tests; no runnable services" OFF)
option(ASTRA_MEASURE_ALLOCATIONS "Instrument C++ new in separate measurement binaries only" OFF)
option(ASTRA_BUILD_BENCHMARKS "Build isolated transport probes; no production protocol extension" OFF)
set(ASTRA_SANITIZER "" CACHE STRING "Optional address,undefined or thread sanitizer")
set_property(CACHE ASTRA_SANITIZER PROPERTY STRINGS "" address,undefined thread)
set(ASTRA_CONTRACT_SEMANTIC "enforce" CACHE STRING "Internal diagnostic comparison only: enforce or ignore")
set_property(CACHE ASTRA_CONTRACT_SEMANTIC PROPERTY STRINGS enforce ignore)
if(NOT ASTRA_CONTRACT_SEMANTIC MATCHES "^(enforce|ignore)$")
    message(FATAL_ERROR "Contract semantic must be enforce or ignore; assume is not supported")
endif()
if(ASTRA_SANITIZER AND NOT ASTRA_SANITIZER MATCHES "^(address,undefined|thread)$")
    message(FATAL_ERROR "Sanitizer must be empty, address,undefined or thread")
endif()
# 所有组合在 core-only 的提前返回前检查, 不能接受请求后静默省略目标.
if(ASTRA_CORE_ONLY AND (ASTRA_MEASURE_ALLOCATIONS OR ASTRA_BUILD_BENCHMARKS))
    message(FATAL_ERROR "Core-only mode has no service allocation or benchmark targets")
endif()
if(ASTRA_MEASURE_ALLOCATIONS AND (ASTRA_SANITIZER OR NOT CMAKE_BUILD_TYPE STREQUAL "Release"))
    message(FATAL_ERROR "Allocation measurement requires a separate Release build without sanitizers")
endif()

# 准备步骤另行授权且显式执行; 正常配置只消费指定前缀, 绝不安装或获取依赖.
# gRPC 导出包会再次 find_dependency(ZLIB), 消费端也必须选择静态库, 不能只在依赖构建时设置.
set(ZLIB_USE_STATIC_LIBS ON)

find_package(yyjson CONFIG REQUIRED)
# yyjson 上游只安装导出目标, 没有 package-version 文件. 从该目标实际导出的头文件校验固定版本.
get_target_property(star_yyjson_includes yyjson::yyjson INTERFACE_INCLUDE_DIRECTORIES)
find_path(star_yyjson_headers yyjson.h PATHS ${star_yyjson_includes} NO_DEFAULT_PATH REQUIRED)
file(STRINGS "${star_yyjson_headers}/yyjson.h" star_yyjson_version REGEX "^#define YYJSON_VERSION_STRING")
if(NOT star_yyjson_version MATCHES "\"0\\.13\\.0\"")
    message(FATAL_ERROR "Star/Planet requires the prepared yyjson 0.13.0 headers and library")
endif()
set(wire_dir "${CMAKE_CURRENT_SOURCE_DIR}/common/src/generated")
foreach(schema IN ITEMS astra orbit comet pulsar polaris)
    foreach(suffix IN ITEMS pb.cc pb.h grpc.pb.cc grpc.pb.h)
        set(generated_file "${wire_dir}/${schema}.${suffix}")
        if(NOT EXISTS "${generated_file}")
            message(FATAL_ERROR "Missing generated protocol: ${generated_file}. Run the explicit project proto command first.")
        endif()
        file(SIZE "${generated_file}" generated_size)
        if(generated_size EQUAL 0)
            message(FATAL_ERROR "Empty generated protocol: ${generated_file}. Generate real sources; placeholders are not accepted.")
        endif()
    endforeach()
endforeach()
add_library(star_wire STATIC
    "${wire_dir}/astra.pb.cc" "${wire_dir}/orbit.pb.cc" "${wire_dir}/comet.pb.cc" "${wire_dir}/pulsar.pb.cc" "${wire_dir}/polaris.pb.cc"
    "${wire_dir}/astra.grpc.pb.cc" "${wire_dir}/orbit.grpc.pb.cc" "${wire_dir}/comet.grpc.pb.cc" "${wire_dir}/pulsar.grpc.pb.cc" "${wire_dir}/polaris.grpc.pb.cc")
target_include_directories(star_wire SYSTEM PUBLIC "${wire_dir}")
target_link_libraries(star_wire PUBLIC protobuf::libprotobuf gRPC::grpc++)
if(ASTRA_SANITIZER)
    # 生成消息同样插桩, 但不施加手写代码的警告/反射规则.
    target_compile_options(star_wire PRIVATE "-fsanitize=${ASTRA_SANITIZER}" -fno-omit-frame-pointer)
endif()
if(ASTRA_SANITIZER MATCHES "address")
    # Protobuf 36.1 默认在 ASan 下改变 ClassData 布局, 必须与既有普通预编译库保持 ABI 一致.
    # 上游兼容开关只固定元数据布局, 不关闭 ASan/UBSan 或 has-bit 一致性检查; 传播给全部协议消费者.
    target_compile_definitions(star_wire PUBLIC PROTOBUF_MESSAGE_GLOBALS_TEMPORARY_OPTOUT=1)
endif()
