cmake_minimum_required(VERSION 3.28)
project(astra_star_cpp VERSION 0.1.0 LANGUAGES C CXX)

# 首版仅支持项目内 GCC 16.2 / Linux, 缺少能力时明确失败, 不降级标准或下载替代工具.
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_CXX_COMPILER_ID STREQUAL "GNU" OR NOT CMAKE_CXX_COMPILER_VERSION VERSION_EQUAL 16.2.0)
    message(FATAL_ERROR "Star/Planet C++ requires Linux and the pinned GCC 16.2.0 toolchain")
endif()
set(CMAKE_CXX_STANDARD 26)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
# 开发产物可直接运行. 安装阶段不继承这个开发机路径, 发布时另行提供匹配运行库.
execute_process(COMMAND "${CMAKE_CXX_COMPILER}" -print-file-name=libstdc++.so.6
    OUTPUT_VARIABLE star_cxx_runtime OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
get_filename_component(star_cxx_runtime_dir "${star_cxx_runtime}" DIRECTORY)
set(CMAKE_BUILD_RPATH "${star_cxx_runtime_dir}")
set(FETCHCONTENT_FULLY_DISCONNECTED ON CACHE BOOL "No implicit downloads" FORCE)
option(ASTRA_CORE_ONLY "Build only offline configuration and role-state tests; no runnable services" OFF)
option(ASTRA_MEASURE_ALLOCATIONS "Instrument C++ new in separate measurement binaries only" OFF)
option(ASTRA_BUILD_BENCHMARKS "Build isolated transport probes; no production protocol extension" OFF)
set(ASTRA_SANITIZER "" CACHE STRING "Optional address,undefined or thread sanitizer")
set_property(CACHE ASTRA_SANITIZER PROPERTY STRINGS "" address,undefined thread)
set(ASTRA_CONTRACT_SEMANTIC "enforce" CACHE STRING "Internal diagnostic comparison only: enforce or ignore")
set_property(CACHE ASTRA_CONTRACT_SEMANTIC PROPERTY STRINGS enforce ignore)
if(NOT ASTRA_CONTRACT_SEMANTIC MATCHES "^(enforce|ignore)$")
    message(FATAL_ERROR "Contract semantic must be enforce or ignore; assume is not supported")
endif()
if(ASTRA_SANITIZER AND NOT ASTRA_SANITIZER MATCHES "^(address,undefined|thread)$")
    message(FATAL_ERROR "Sanitizer must be empty, address,undefined or thread")
endif()
# 所有组合在 core-only 的提前返回前检查, 不能接受请求后静默省略目标.
if(ASTRA_CORE_ONLY AND (ASTRA_MEASURE_ALLOCATIONS OR ASTRA_BUILD_BENCHMARKS))
    message(FATAL_ERROR "Core-only mode has no service allocation or benchmark targets")
endif()
if(ASTRA_MEASURE_ALLOCATIONS AND (ASTRA_SANITIZER OR NOT CMAKE_BUILD_TYPE STREQUAL "Release"))
    message(FATAL_ERROR "Allocation measurement requires a separate Release build without sanitizers")
endif()

include(CTest)
find_package(Threads REQUIRED)
add_library(star_options INTERFACE)
target_compile_options(star_options INTERFACE -Wall -Wextra -Wpedantic -Werror -Wconversion -Wsign-conversion -freflection -fcontracts
    "-fcontract-evaluation-semantic=${ASTRA_CONTRACT_SEMANTIC}")
target_compile_definitions(star_options INTERFACE _GLIBCXX_ASSERTIONS=1)
# GCC 的契约违例处理位于附带的静态扩展库, 必须随启用契约的目标一起链接.
target_link_libraries(star_options INTERFACE stdc++exp)
if(ASTRA_SANITIZER)
    target_compile_options(star_options INTERFACE "-fsanitize=${ASTRA_SANITIZER}" -fno-omit-frame-pointer)
    target_link_options(star_options INTERFACE "-fsanitize=${ASTRA_SANITIZER}")
endif()

# 生成协议和第三方目标不继承手写代码的反射、契约或警告策略.
add_library(star_core STATIC common/src/types.cpp common/src/config.cpp common/src/policy.cpp)
target_include_directories(star_core PUBLIC common/include)
target_link_libraries(star_core PUBLIC star_options Threads::Threads)
add_library(star_star STATIC star/src/topology.cpp)
target_include_directories(star_star PUBLIC star/src)
target_link_libraries(star_star PUBLIC star_core)
add_library(star_planet STATIC planet/src/upstream.cpp)
target_include_directories(star_planet PUBLIC planet/src)
target_link_libraries(star_planet PUBLIC star_core)

# 存储工具和分配失败测试不链接 gRPC, 也可在 core-only 模式独立编译.
add_library(star_sync_store STATIC common/src/store.cpp)
target_include_directories(star_sync_store PUBLIC common/src common/include)
target_link_libraries(star_sync_store PUBLIC star_options Threads::Threads)

if(BUILD_TESTING)
    find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
    add_test(NAME cpp_build COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_SOURCE_DIR}/test_build.py")
    set_tests_properties(cpp_build PROPERTIES TIMEOUT 30)
    add_executable(star_core_test common/tests/core_test.cpp)
    target_link_libraries(star_core_test PRIVATE star_star star_planet)
    add_test(NAME cpp_core COMMAND star_core_test)
    set_tests_properties(cpp_core PROPERTIES TIMEOUT 20)
    add_executable(star_contracts_probe common/tests/contracts_probe.cpp)
    target_link_libraries(star_contracts_probe PRIVATE star_options)
    add_test(NAME cpp_language COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_SOURCE_DIR}/test_language.py"
        --compiler "${CMAKE_CXX_COMPILER}" --probe "$<TARGET_FILE:star_contracts_probe>"
        --semantic "${ASTRA_CONTRACT_SEMANTIC}" --temporary-root "${CMAKE_CURRENT_BINARY_DIR}")
    set_tests_properties(cpp_language PROPERTIES TIMEOUT 180)
    foreach(suite IN ITEMS store store_fault store_ttl)
        add_executable("star_${suite}_test" "common/tests/${suite}_test.cpp")
        target_link_libraries("star_${suite}_test" PRIVATE star_sync_store)
        add_test(NAME "cpp_${suite}" COMMAND "star_${suite}_test")
        set_tests_properties("cpp_${suite}" PROPERTIES TIMEOUT 30)
    endforeach()
    # 时间轮是独立头文件工具, 不链接服务或存储; Release 和 sanitizer 使用相同断言与测试入口.
    foreach(suite IN ITEMS wheel wheel_allocation)
        add_executable("star_${suite}_test" "common/tests/${suite}_test.cpp")
        target_include_directories("star_${suite}_test" PRIVATE common/src)
        target_link_libraries("star_${suite}_test" PRIVATE star_options)
        add_test(NAME "cpp_${suite}" COMMAND "star_${suite}_test")
        set_tests_properties("cpp_${suite}" PROPERTIES TIMEOUT 60)
    endforeach()
endif()

if(ASTRA_CORE_ONLY)
    message(STATUS "Core-only scope: gRPC services and runtime qualification are not built")
    return()
endif()

# 准备步骤另行授权且显式执行; 正常配置只消费指定前缀, 绝不安装或获取依赖.
# gRPC 导出包会再次 find_dependency(ZLIB), 消费端也必须选择静态库, 不能只在依赖构建时设置.
set(ZLIB_USE_STATIC_LIBS ON)
find_package(Protobuf 36.1.0 EXACT CONFIG REQUIRED)
find_package(gRPC 1.84.0 EXACT CONFIG REQUIRED)
find_package(yyjson CONFIG REQUIRED)
# yyjson 上游只安装导出目标, 没有 package-version 文件. 从该目标实际导出的头文件校验固定版本.
get_target_property(star_yyjson_includes yyjson::yyjson INTERFACE_INCLUDE_DIRECTORIES)
find_path(star_yyjson_headers yyjson.h PATHS ${star_yyjson_includes} NO_DEFAULT_PATH REQUIRED)
file(STRINGS "${star_yyjson_headers}/yyjson.h" star_yyjson_version REGEX "^#define YYJSON_VERSION_STRING")
if(NOT star_yyjson_version MATCHES "\"0\\.13\\.0\"")
    message(FATAL_ERROR "Star/Planet requires the prepared yyjson 0.13.0 headers and library")
endif()
set(wire_dir "${CMAKE_CURRENT_SOURCE_DIR}/common/src/generated")
foreach(schema IN ITEMS astra orbit comet)
    foreach(suffix IN ITEMS pb.cc pb.h grpc.pb.cc grpc.pb.h)
        set(generated_file "${wire_dir}/${schema}.${suffix}")
        if(NOT EXISTS "${generated_file}")
            message(FATAL_ERROR "Missing generated protocol: ${generated_file}. Run the explicit project proto command first.")
        endif()
        file(SIZE "${generated_file}" generated_size)
        if(generated_size EQUAL 0)
            message(FATAL_ERROR "Empty generated protocol: ${generated_file}. Generate real sources; placeholders are not accepted.")
        endif()
    endforeach()
endforeach()
add_library(star_wire STATIC
    "${wire_dir}/astra.pb.cc" "${wire_dir}/orbit.pb.cc" "${wire_dir}/comet.pb.cc"
    "${wire_dir}/astra.grpc.pb.cc" "${wire_dir}/orbit.grpc.pb.cc")
target_include_directories(star_wire SYSTEM PUBLIC "${wire_dir}")
target_link_libraries(star_wire PUBLIC protobuf::libprotobuf gRPC::grpc++)
if(ASTRA_SANITIZER STREQUAL "thread")
    # TSan 需要看到生成消息跨线程交接时的访问, 但仍不对生成代码施加手写警告/反射规则.
    target_compile_options(star_wire PRIVATE -fsanitize=thread -fno-omit-frame-pointer)
endif()

# BoringSSL 来自 gRPC 的 module provider, 与 TLS 和私有身份适配共用一份库.
if(NOT TARGET gRPC::Crypto)
    message(FATAL_ERROR "The prepared gRPC package must export its paired BoringSSL crypto target")
endif()
add_library(star_runtime STATIC common/src/identity.cpp common/src/admission.cpp common/src/grpc_session.cpp common/src/process.cpp common/src/runtime.cpp)
target_include_directories(star_runtime PRIVATE common/src)
target_link_libraries(star_runtime PUBLIC star_core PRIVATE star_wire gRPC::Crypto yyjson::yyjson Threads::Threads)
if(BUILD_TESTING)
    foreach(suite IN ITEMS identity session admission connect process proto_proxy)
        add_executable("star_${suite}_test" "common/tests/${suite}_test.cpp")
        target_include_directories("star_${suite}_test" PRIVATE common/src)
        target_compile_definitions("star_${suite}_test" PRIVATE ASTRA_FIXTURES="${CMAKE_CURRENT_SOURCE_DIR}/../cluster/tests/fixtures")
        target_link_libraries("star_${suite}_test" PRIVATE star_runtime star_wire gRPC::Crypto)
        add_test(NAME "cpp_${suite}" COMMAND "star_${suite}_test")
        set_tests_properties("cpp_${suite}" PROPERTIES TIMEOUT 30)
    endforeach()
    target_link_libraries(star_identity_test PRIVATE yyjson::yyjson)
    add_executable(star_rpc_probe common/tests/rpc_probe.cpp)
    target_include_directories(star_rpc_probe PRIVATE common/src)
    target_link_libraries(star_rpc_probe PRIVATE star_runtime star_wire)
endif()
add_executable(star star/src/main.cpp)
target_link_libraries(star PRIVATE star_star star_runtime)
add_executable(planet planet/src/main.cpp)
target_link_libraries(planet PRIVATE star_planet star_runtime)
if(ASTRA_BUILD_BENCHMARKS)
    # 存储微基准独立链接内部模块, 不启动网络服务或读取业务配置.
    add_executable(star_sync_store_bench bench/store.cpp)
    target_link_libraries(star_sync_store_bench PRIVATE star_sync_store)
    add_library(star_bench_wire STATIC bench/generated/probe.pb.cc bench/generated/probe.grpc.pb.cc)
    target_include_directories(star_bench_wire SYSTEM PUBLIC bench/generated)
    target_link_libraries(star_bench_wire PUBLIC protobuf::libprotobuf gRPC::grpc++)
    if(ASTRA_SANITIZER STREQUAL "thread")
        target_compile_options(star_bench_wire PRIVATE -fsanitize=thread -fno-omit-frame-pointer)
    endif()
    add_executable(star_push_probe bench/push.cpp)
    target_include_directories(star_push_probe PRIVATE common/src common/tests)
    target_compile_definitions(star_push_probe PRIVATE ASTRA_FIXTURES="${CMAKE_CURRENT_SOURCE_DIR}/../cluster/tests/fixtures")
    target_link_libraries(star_push_probe PRIVATE star_runtime star_wire star_bench_wire gRPC::Crypto)
endif()
if(ASTRA_MEASURE_ALLOCATIONS)
    add_library(star_allocation_measure OBJECT common/tests/allocation_measure.cpp)
    target_link_libraries(star_allocation_measure PRIVATE star_options)
    target_sources(star PRIVATE $<TARGET_OBJECTS:star_allocation_measure>)
    target_sources(planet PRIVATE $<TARGET_OBJECTS:star_allocation_measure>)
    if(ASTRA_BUILD_BENCHMARKS)
        target_sources(star_push_probe PRIVATE $<TARGET_OBJECTS:star_allocation_measure>)
    endif()
endif()
install(TARGETS star planet RUNTIME DESTINATION bin)
install(DIRECTORY licenses/ DESTINATION share/astra/licenses)
install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/../LICENSE" DESTINATION share/astra)
