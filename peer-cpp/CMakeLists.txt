cmake_minimum_required(VERSION 3.28)
project(verdandi_peer_cpp VERSION 0.1.0 LANGUAGES C CXX)

# 首版仅支持项目内 GCC 16.2 / Linux, 缺少能力时明确失败, 不降级标准或下载替代工具.
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_CXX_COMPILER_ID STREQUAL "GNU" OR NOT CMAKE_CXX_COMPILER_VERSION VERSION_EQUAL 16.2.0)
    message(FATAL_ERROR "Star/Planet C++ requires Linux and the pinned GCC 16.2.0 toolchain")
endif()
set(CMAKE_CXX_STANDARD 26)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
# 开发产物可直接运行. 安装阶段不继承这个开发机路径, 发布时另行提供匹配运行库.
execute_process(COMMAND "${CMAKE_CXX_COMPILER}" -print-file-name=libstdc++.so.6
    OUTPUT_VARIABLE peer_cxx_runtime OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
get_filename_component(peer_cxx_runtime_dir "${peer_cxx_runtime}" DIRECTORY)
set(CMAKE_BUILD_RPATH "${peer_cxx_runtime_dir}")
set(FETCHCONTENT_FULLY_DISCONNECTED ON CACHE BOOL "No implicit downloads" FORCE)
option(VERDANDI_CORE_ONLY "Build only offline configuration and role-state tests; no runnable services" OFF)
set(VERDANDI_SANITIZER "" CACHE STRING "Optional address,undefined or thread sanitizer")

include(CTest)
find_package(Threads REQUIRED)
add_library(peer_options INTERFACE)
target_compile_options(peer_options INTERFACE -Wall -Wextra -Wpedantic -Werror -Wconversion -Wsign-conversion -freflection -fcontracts
    -fcontract-evaluation-semantic=enforce)
target_compile_definitions(peer_options INTERFACE _GLIBCXX_ASSERTIONS=1)
# GCC 的契约违例处理位于附带的静态扩展库, 必须随启用契约的目标一起链接.
target_link_libraries(peer_options INTERFACE stdc++exp)
if(VERDANDI_SANITIZER)
    target_compile_options(peer_options INTERFACE "-fsanitize=${VERDANDI_SANITIZER}" -fno-omit-frame-pointer)
    target_link_options(peer_options INTERFACE "-fsanitize=${VERDANDI_SANITIZER}")
endif()

# 生成协议和第三方目标不继承手写代码的反射、契约或警告策略.
add_library(peer_core STATIC common/src/types.cpp common/src/config.cpp common/src/policy.cpp)
target_include_directories(peer_core PUBLIC common/include)
target_link_libraries(peer_core PUBLIC peer_options Threads::Threads)
add_library(peer_star STATIC star/src/topology.cpp)
target_include_directories(peer_star PUBLIC star/src)
target_link_libraries(peer_star PUBLIC peer_core)
add_library(peer_planet STATIC planet/src/upstream.cpp)
target_include_directories(peer_planet PUBLIC planet/src)
target_link_libraries(peer_planet PUBLIC peer_core)

if(BUILD_TESTING)
    add_executable(peer_core_test common/tests/core_test.cpp)
    target_link_libraries(peer_core_test PRIVATE peer_star peer_planet)
    add_test(NAME cpp_core COMMAND peer_core_test)
    set_tests_properties(cpp_core PROPERTIES TIMEOUT 20)
endif()

if(VERDANDI_CORE_ONLY)
    message(STATUS "Core-only scope: gRPC services and runtime qualification are not built")
    return()
endif()

# 准备步骤另行授权且显式执行; 正常配置只消费指定前缀, 绝不安装或获取依赖.
# gRPC 导出包会再次 find_dependency(ZLIB), 消费端也必须选择静态库, 不能只在依赖构建时设置.
set(ZLIB_USE_STATIC_LIBS ON)
find_package(Protobuf 36.1.0 EXACT CONFIG REQUIRED)
find_package(gRPC 1.84.0 EXACT CONFIG REQUIRED)
find_package(yyjson CONFIG REQUIRED)
# yyjson 上游只安装导出目标, 没有 package-version 文件. 从该目标实际导出的头文件校验固定版本.
get_target_property(peer_yyjson_includes yyjson::yyjson INTERFACE_INCLUDE_DIRECTORIES)
find_path(peer_yyjson_headers yyjson.h PATHS ${peer_yyjson_includes} NO_DEFAULT_PATH REQUIRED)
file(STRINGS "${peer_yyjson_headers}/yyjson.h" peer_yyjson_version REGEX "^#define YYJSON_VERSION_STRING")
if(NOT peer_yyjson_version MATCHES "\"0\\.13\\.0\"")
    message(FATAL_ERROR "Star/Planet requires the prepared yyjson 0.13.0 headers and library")
endif()
set(wire_dir "${CMAKE_CURRENT_SOURCE_DIR}/common/src/generated")
foreach(schema IN ITEMS peer admission peer_transport)
    if(NOT EXISTS "${wire_dir}/${schema}.pb.cc")
        message(FATAL_ERROR "Missing generated C++ protocol sources. Run the explicit project proto command first.")
    endif()
endforeach()
add_library(peer_wire STATIC
    "${wire_dir}/peer.pb.cc" "${wire_dir}/admission.pb.cc" "${wire_dir}/peer_transport.pb.cc"
    "${wire_dir}/admission.grpc.pb.cc" "${wire_dir}/peer_transport.grpc.pb.cc")
target_include_directories(peer_wire SYSTEM PUBLIC "${wire_dir}")
target_link_libraries(peer_wire PUBLIC protobuf::libprotobuf gRPC::grpc++)
if(VERDANDI_SANITIZER STREQUAL "thread")
    # TSan 需要看到生成消息跨线程交接时的访问, 但仍不对生成代码施加手写警告/反射规则.
    target_compile_options(peer_wire PRIVATE -fsanitize=thread -fno-omit-frame-pointer)
endif()

# BoringSSL 来自 gRPC 的 module provider, 与 TLS 和私有身份适配共用一份库.
if(NOT TARGET gRPC::Crypto)
    message(FATAL_ERROR "The prepared gRPC package must export its paired BoringSSL crypto target")
endif()
add_library(peer_runtime STATIC common/src/identity.cpp common/src/admission.cpp common/src/grpc_session.cpp common/src/runtime.cpp)
target_include_directories(peer_runtime PRIVATE common/src)
target_link_libraries(peer_runtime PUBLIC peer_core PRIVATE peer_wire gRPC::Crypto yyjson::yyjson Threads::Threads)
if(BUILD_TESTING)
    foreach(suite IN ITEMS identity session admission connect)
        add_executable("peer_${suite}_test" "common/tests/${suite}_test.cpp")
        target_include_directories("peer_${suite}_test" PRIVATE common/src)
        target_compile_definitions("peer_${suite}_test" PRIVATE VERDANDI_FIXTURES="${CMAKE_CURRENT_SOURCE_DIR}/../peer/tests/fixtures")
        target_link_libraries("peer_${suite}_test" PRIVATE peer_runtime peer_wire gRPC::Crypto)
        add_test(NAME "cpp_${suite}" COMMAND "peer_${suite}_test")
        set_tests_properties("cpp_${suite}" PROPERTIES TIMEOUT 30)
    endforeach()
    target_link_libraries(peer_identity_test PRIVATE yyjson::yyjson)
    add_executable(peer_rpc_probe common/tests/rpc_probe.cpp)
    target_include_directories(peer_rpc_probe PRIVATE common/src)
    target_link_libraries(peer_rpc_probe PRIVATE peer_runtime peer_wire)
endif()
add_executable(peer star/src/main.cpp)
target_link_libraries(peer PRIVATE peer_star peer_runtime)
add_executable(planet planet/src/main.cpp)
target_link_libraries(planet PRIVATE peer_planet peer_runtime)
install(TARGETS peer planet RUNTIME DESTINATION bin)
install(DIRECTORY licenses/ DESTINATION share/verdandi/licenses)
install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/../LICENSE" DESTINATION share/verdandi)
